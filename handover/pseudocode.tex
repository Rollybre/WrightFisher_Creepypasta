\subsection{Frequencies}

Frequencies are computed \emph{synchronously}, once per generation, over all
individuals created so far (no death, uniform memory, $\tau = \infty$):
\begin{equation}
  f_{k,t} \;=\; \frac{\sum_{i \in \mathcal{P}_t} \mathbb{1}[k \in \mathbf{x}_i]}
                     {\sum_{i \in \mathcal{P}_t} |\mathbf{x}_i|},
  \qquad k = 1,\dots,C_t,
  \label{eq:freq}
\end{equation}
where $\mathcal{P}_t$ is the set of all individuals existing at the start of
generation $t$.

\subsection{Number of tags}

\section{Pseudo-code}
% ---------------------------------------------------------------------------

\begin{algorithm}[h]
  \caption{One simulation run.}
  \label{alg:sim}
  \KwParams{$\mu$, $\lambda$, $\mathbf{q} = (q_1,\dots,q_R)$, $C_0$,n}
  \KwOut{Vector of tags  $\{\mathbf{x}_i\}$ for all individuals}
  Initialise population $N_i$ with $C \leftarrow C_0$ classes\;
  \For{$t = 1$ \KwTo $T$}{
    Compute frequency of tag $f_x$ based on actual population\;
    $n_t = \text{popgrwoth}(t)$\;
    \For{$i = 1$ \KwTo $n_t$}{ 
      Draw $m_i \sim \Pois(\lambda)$ truncated to $m_i \ge 1$\;
      $S \leftarrow \emptyset$\;
      \For{$r = 1$ \KwTo $m_i$}{
          $q \leftarrow q_{\min(r,R)}$ \;
          Compute weight of all tag usiq $q_r$ and $f$\;
          \lIf{$\sum_k w_k = 0$}{\textbf{break}}
          select a tag $k \sim w$; $S \leftarrow S \cup \{k\}$ \Comment*[r]{conformist copy}
      }
      $C' \leftarrow \text{GenerateNewTags}(n_t,t)$\Comment*[r]{innovation}
      Randomly add the $C'$ tag to $C'$ existing stories\;
      $\mathbf{x}_i \leftarrow S$\;
    }
    $\mathcal{P}_t \leftarrow \mathcal{P}_{t-1} \cup \{\mathbf{x}_1,\dots,\mathbf{x}_{n_t}\}$\;
  }
\end{algorithm}
